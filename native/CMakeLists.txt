cmake_minimum_required(VERSION 3.20)
project(thug2_core LANGUAGES CXX)
add_library(thug2_core src/core.cpp src/assets.cpp src/qb.cpp src/precompiled.cpp src/symbols.cpp src/values.cpp src/runtime.cpp src/wait.cpp src/random.cpp)
set_target_properties(thug2_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(thug2_core PUBLIC include)
target_compile_features(thug2_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(thug2_core PRIVATE /W4)
else()
  target_compile_options(thug2_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(thug2-prx src/main.cpp)
target_link_libraries(thug2-prx PRIVATE thug2_core)
add_executable(thug2-assets src/assets_main.cpp)
target_link_libraries(thug2-assets PRIVATE thug2_core)
add_executable(thug2-qb src/qb_main.cpp)
target_link_libraries(thug2-qb PRIVATE thug2_core)
add_executable(thug2-symbols src/symbols_main.cpp)
target_link_libraries(thug2-symbols PRIVATE thug2_core)
add_executable(thug2-run src/runtime_main.cpp)
target_link_libraries(thug2-run PRIVATE thug2_core)
include(CTest)
